// ============= Syncronizing Data Exchange
func main() {

	data := make(chan string)

	go func() {
		for idx := range 5 {
			idxs := string('0' + idx)
			data <- "hello " + idxs
		}
		time.Sleep(100 * time.Millisecond)
		close(data)
	}()

	for value := range data {
		fmt.Println("Recieved value:", value, ":", time.Now())
	} // loops over only on active channel, creates reciever
	//each time and stops creating reciever (looping) once the
	//channel is closed

}
