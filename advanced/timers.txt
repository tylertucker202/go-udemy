package main

import (
	"fmt"
	"time"
)

func main() {
	timer1 := time.NewTimer(1 * time.Second) //non blocking timer starts
	timer2 := time.NewTimer(2 * time.Second) //non blocking timer starts

	for range 2 {
		select {
		case <-timer1.C:
			fmt.Println("Timer1 expired")
		case <-timer2.C:
			fmt.Println("Timer2 expired")
		}
	}
}
