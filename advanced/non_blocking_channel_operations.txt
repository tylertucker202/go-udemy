	// === NON BLOCKING OPERATION IN REAL TIME SYSTEMS

	data := make(chan int)

	quit := make(chan bool)

	go func() {
	outer:
		for {
			select {
			case d := <-data:
				fmt.Println("Data received:", d)
			case <-quit:
				fmt.Println("Stopping...")
				break outer
			default:
				fmt.Println("Waiting for data...")
				time.Sleep(500 * time.Millisecond)
			}
		}
	}()

	for i := 0; i < 5; i++ {
		data <- i
		time.Sleep(1000 * time.Millisecond)
	}
	quit <- true

}
