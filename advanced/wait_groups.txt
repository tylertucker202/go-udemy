package main

import (
	"fmt"
	"sync"
	"time"
)

// TICKETS EXAMPLE
type ticketRequest struct {
	personID   int
	numTickets int
	cost       int
}

func (t *ticketRequest) ProcessTicket(wg *sync.WaitGroup) {
	defer wg.Done()
	fmt.Printf("Processing %d ticket(s) of personID %d with total cost %d\n", t.numTickets, t.personID, t.cost)
	time.Sleep(time.Second)
	fmt.Printf("Ticket Processed for id: %d cost: %d\n", t.personID, t.cost)
}

func main() {
	var wg sync.WaitGroup
	numRequests := 10
	price := 10
	// ticketRequests := make(chan ticketRequest, numRequests)

	for i := range numRequests {
		tr := ticketRequest{personID: i + 1, numTickets: (i + 1) * 2, cost: (i + 1) * price}
		wg.Add(1)
		tr.ProcessTicket(&wg)
	}

	// Wait for all tickets to process
	wg.Wait()

	//finished
	fmt.Println("All tickets processed")

}
