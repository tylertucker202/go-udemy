package main

import (
	"fmt"
	"math"
)

type geometry interface {
	area() float64
	perim() float64
}

type rect struct {
	width, height float64
}

type rect1 struct {
	width, height float64
}

type circle struct {
	radius float64
}

func (r rect) area() float64 {
	return r.width * r.height
}

func (r rect) perim() float64 {
	return 2 * (r.width + r.height)
}

func (r rect1) area() float64 {
	return r.width * r.height
}

func (c circle) area() float64 {
	return math.Pi * c.radius * c.radius
}

// Note: circle can have a diameter method and still use the geometry interface, as long as it implements the area() and perim() methods.
// The diameter method is an additional method that does not affect the implementation of the geometry interface.
func (c circle) diameter() float64 {
	return 2 * c.radius
}

func (c circle) perim() float64 {
	return 2 * math.Pi * c.radius
}

func measure(g geometry) {
	fmt.Println(g)
	fmt.Printf("area: %.2f\n", g.area())
	fmt.Printf("perimeter: %.2f\n", g.perim())
}

func main() {

	r := rect{width: 3, height: 4}
	c := circle{radius: 5}

	measure(r)
	measure(c)
	// This will cause a compile-time error because rect1 does not implement the geometry interface (it lacks the perim() method).
	// measure(rect1{width: 3, height: 4})

	myPrinter("Hello", "World", 123, 45.67)
	printType("Hello")
	printType(123)
	printType(45.67)

}

func printType(i interface{}) {
	switch v := i.(type) {
	case string:
		fmt.Printf("The type of the value is string: %s\n", v)
	case int:
		fmt.Printf("The type of the value is int: %d\n", v)
	case float64:
		fmt.Printf("The type of the value is float64: %.2f\n", v)
	default:
		fmt.Printf("Unknown type: %T\n", v)
	}
}

// myPrinter is a variadic function that takes a variable number of arguments of any type (interface{}).
// It iterates over the arguments and prints each one on a new line using fmt.Println.
func myPrinter(i ...interface{}) {
	for _, v := range i {
		fmt.Println(v)
	}
}
