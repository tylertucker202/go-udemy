func checkType(x interface{}) {
	switch x.(type) {
	case int:
		println("It's an integer.")
	case string:
		println("It's a string.")
	case bool:
		println("It's a boolean.")
	case float64:
		println("It's a float.")
	default:
		println("Unknown type.")
	}
}
