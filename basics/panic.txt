package main

import "fmt"

func main() {

	// panic (interface{})

	// Example of a valid input
	process(10)

	// Example of an invalid input that will cause a panic
	process(-5)

}

func process(input int) {

	defer fmt.Println("deferred 1")
	defer fmt.Println("deferred 2")
	if input < 0 {
		fmt.Println("Input is negative, about to panic")
		panic(fmt.Sprintf("Negative input: %d", input))
		// fmt.Println("This line will not be executed due to panic")
		// defer fmt.Println("deferred 3") // This will not be executed
	}
	fmt.Printf("Processing input: %d\n", input)
}
